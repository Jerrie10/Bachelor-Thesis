\begin{frame}
  \frametitle{Access function}
  \begin{columns}
    \begin{column}{0.68\textwidth}
      \begin{align*}
        \text{Access}(y) =& \sum_{\substack{\mathcal{K} \text{ minimum elements of }\\ {\{A_i(y)| i\in \tilde V^+\}}}} A_i(y)\\
        A_i(y) =& \sum_{j\in\tilde V^-} \frac{S_j d_{ij}^{-\beta}(y)}{F_j(y)}  \hspace{4mm} i\in V^+\\
        S_j, F_j(y) :& \text{ destination attributes}\\
        d_{ij}(y):& \text{ travel time from $i$ to $j$}\\
        \beta >&0 \text{, gravity decay parameter}
      \end{align*}
